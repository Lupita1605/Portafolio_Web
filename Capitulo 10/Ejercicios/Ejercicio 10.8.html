10.8 Escriba instrucciones de JavaScript para realizar cada una de las siguientes tareas:

a) Mostrar el valor del séptimo elemento del arreglo f.

document.writeln( f[ 6 ] );
b) Inicializar cada uno de los cinco elementos del arreglo unidimensional g en 8.
var g = [ 8, 8, 8, 8, 8 ];

// O mediante un ciclo:
for ( var i = 0; i < g.length; ++i )
{
   g[ i ] = 8;
}
c) Calcular el total de los elementos del arreglo c, que contiene 100 elementos numéricos.
var total = 0;
for ( var i = 0; i < c.length; ++i )
{
   total += c[ i ];
}
d) Copiar el arreglo a de 11 elementos en la primera porción del arreglo b, que contiene 34 elementos.
for ( var i = 0; i < a.length; ++i )
{
   b[ i ] = a[ i ];
}
e) Determinar e imprimir el valor más pequeño y el más grande contenido en el arreglo w de 99 elementos de punto flotante.
var menor = w[ 0 ];
var mayor = w[ 0 ];

for ( var i = 1; i < w.length; ++i )
{
   if ( w[ i ] < menor )
      menor = w[ i ];

   if ( w[ i ] > mayor )
      mayor = w[ i ];
}

document.writeln( "Menor: " + menor + ", Mayor: " + mayor );
